	name: 	""

	width:	vedi Dim2D
	height:	vedi Dim2D
	
	NB: sameAsSwapchain è garantito essere una delle due seguenti (dipende dall'HW)
			VK_FORMAT_R8G8B8A8_SRGB
			VK_FORMAT_B8G8R8A8_SRGB
}

depthStencil
{
	name: ""
	format:	best | eImageFormat
	width:	vedi Dim2D
	height:	vedi Dim2D
	stencil: 0|1		default 0
	
	NB: "best" è garantito essere una delle due seguenti, in ordine di priorità (dipende dall'HW):
			VK_FORMAT_D32_SFLOAT
			VK_FORMAT_D16_UNORM
			
		oppure (se richiesto anche lo stencil):
			VK_FORMAT_D32_SFLOAT_S8_UINT
			VK_FORMAT_D24_UNORM_S8_UINT
			VK_FORMAT_D16_UNORM_S8_UINT
}
