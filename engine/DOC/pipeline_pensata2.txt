def-pipeline
{
	name: line3D
	
	output
	{
		rt: sameAsSwapchain, color_attachment_optimal, load, store
	
		depthStencil: best, depth_attachment_optimal, load, store
	}
	
	zOption
	{
		enabled: 1
		write: 0
		cmpFN: eZFunc::LESSEQUAL
	}

	stencilOption
	{
		enabled: 0
	}

	
	vtxFormat: vtxFormat_0
	cullMode: eCullMode::CCW
	drawPrimitive: eDrawPrimitive::trisList

	vtxShader: lineRenderer.vert.shader
	pxlShader: lineRenderer.frag.shader
	
	descriptor: "@texture_and_sampler"
	descriptor: "@scene"
	descriptor
	{
		name: perInstanceData
		type: dynamic
		add_storageBuffer: VK_SHADER_STAGE_VERTEX_BIT		
	}
	
	push_constant: name, vtx-or-frag, size
}

chain
{
	//risorse che vengono create da questa chain e sono 
	//accessibili da tutti gli step di questa chain.
	//In generale sono istanze di una qualche def-xx definita in precedenza
	resources
	{
		create-rt: <res-name>, <format>, <width>, <height>
		create-depthStencil: <res-name>, <format>, <width>, <height>, <withStencil>
		create-descriptor: <res-name>, <name-of-def-descriptor>
		create-pipeline: <res-name>, <name-of-def-pipeline>
	}
	
	step
	{
		//use-pipeline: <res-name>
		//with-output: <res-name>
		//with-descriptor: <res-name>

		enabled: 1
		use-pipline: pipe-line2D
		with-output: "rt-name"
		with-output: "zb-name"
		with-descriptor: "descr1-name"
		with-descriptor: "descr2-name"			
	}
	
	step
	{
		enabled: 1
		use-pipeline: line3D
		
		with-output
		{
			rt: colorRT
			depthStencil: mainZB
		}
	}	
	
	step
	{
		enabled: 1
		use-pipeline: present_pass
		
		with-output
		{
			rt: colorRT
			depthStencil: mainZB
		}
	}	
}
