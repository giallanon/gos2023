@framebuffer_def
{
	//color e zbuffer (massimo 16)
	colbuf: <ImgFormat = sameAsSwapchain | U8_RGBA_sRGB | ... >, <initialLayout>, <finalLayout>, <loadOp = load | clear | dont_care>, <storeOp = store | dont_care | none>
	...
	colbuf:

	//max 1 zb
	(optional)	zb: <ImgFormat = sameAsSwapchain | DEPTH_F32 | ... > <initialLayout>, <finalLayout>, <loadOp = load | clear | dont_care>, <storeOp = store | dont_care | none>
}



@render_pass
{
	(mandatory)	@framebuffer_def:

	(optional)	clearcol: <attachment-index-0toN>, <#aarrggbb = #f000>
	...

	(optional)	clearzb: <depth = 1> , <stencil = 0>


	//mandatorio almeno 1 sub-pass, massimo 8 subpass
	subpass
	{
		(mandatory)	type: gfx | compute
		(optional)	rt: index of 1 or more <colbuf> definiti in @framebuffer_def
		(optional)	zb: => non serve dato che nel framebuffer_def, ci può essere al max 1 ZB e se nelle zOption qui sotto dico che è enabled, lo so già da me quale è l'indice del target da usare
		(optional)	input: index of 1 or more buf_ definiti in @framebuffer_def. Al posto di index, si può usare "zb" per indicare lo zbuffer
		(optional)	preserve: index of 1 or more buf_ definiti in @framebuffer_def

		(optional)	zOption: 1, 1, LESS				=> <enabled>, <write>, <cmpFN>		=> default: 1, 1, LESS
		(optional)	stencilOption: 0, NEVER			=> <enabled>, <cmpFN>				=> default: 0, NEVER
		(optional)	cullMode: CCW					=> <mode>							=> default: eCullMode::CCW
		(optional)	drawPrimitive: trisList			=> <primitive>						=> default: trisList

		(optional)	@vtx_shader
		(optional)	@pxl_shader
	}
}
